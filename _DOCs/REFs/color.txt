#include <iostream>
#include <string>
#include <chrono>
#include <thread>
#include <vector>
#include <random>

#define _clear_all std::cout << "\033[2J\033[H" << std::flush;
#define _SLEEP_SEC(x) std::this_thread::sleep_for(std::chrono::seconds(x))
#define FOR(x, y) for (int j = 0; j < y; j++) std::cout << x

namespace loging {
    enum color {
        RED,
        GREEN,
        BLUE,
        YELLOW,
        ORANGE,
        WHITE
    };

    std::string get_start(color col) {
        std::string val;
        switch (col) {
            case RED:    val = "\033[1;31m"; break;
            case GREEN:  val = "\033[1;32m"; break;
            case BLUE:   val = "\033[1;34m"; break;
            case YELLOW: val = "\033[1;33m"; break;
            case ORANGE: val = "\033[1;35m"; break;
            case WHITE:
            default:     val = ""; break;
        }
        return val;
    }

    inline std::string get_end() {
        return "\033[0m";
    }
}

template <typename T>
using _VEC = std::vector<T>;

constexpr int lines = 30;
constexpr int wed_MAX = 80;
constexpr int wed_MIN = 30;

using namespace std;
using namespace loging;

int get_random(int min, int max) {
    static std::random_device rd;
    static std::mt19937 gen(rd());
    std::uniform_int_distribution<> distrib(min, max);
    return distrib(gen);
}

void LOG(const string& str, color col = color::WHITE) {
    cout << get_start(col) << str << get_end() << endl;
}

int main() {
    _VEC<color> vec = {color::RED, color::GREEN, color::BLUE,
                       color::YELLOW, color::ORANGE};

    for (;;) {
        int ind = 0;
        for (int i = 0; i < lines; i++) {
            cout << get_start(vec[ind++]);
            FOR(".", get_random(wed_MIN, wed_MAX));
            cout << get_end() << endl;
            if(ind>=vec.size())ind=0;
        }
        //_SLEEP_SEC(1);
        _clear_all;
    }

    return 0;
}
