cmake_minimum_required(VERSION 3.15)
project(game_emulator)

set(CMAKE_CXX_STANDARD 17)

# Collect all cpp files
file(GLOB_RECURSE CONTROLLER_SRC controller/SRCs/*.cpp)
file(GLOB_RECURSE INPUT_SRC     input/SRCs/*.cpp)

add_executable(game_emulator
    game_emulator.cpp
    ${CONTROLLER_SRC}
    ${INPUT_SRC}
)

# Include headers
target_include_directories(game_emulator PRIVATE
    controller/HEADERs
    input/HEADERs
)
